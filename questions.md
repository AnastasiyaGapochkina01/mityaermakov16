```
FROM golang
RUN apt install -y pkg1 pkg2 pkgN # Dependencies for app_
COPY. .
RUN go build -o app main.go
CMD ./app_
```
и
```
FROM ubuntu:18.04 
COPY ./src /app 
RUN apt-get update -y 
RUN apt-get install -y nodejs 
RUN npm install 
ENTRYPOINT ["npm"] 
CMD ["run", "prod"]
```
